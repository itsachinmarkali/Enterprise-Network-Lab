
📄 CORE SWITCH CONFIG (L3 SWITCH)


enable
configure terminal

! VLAN Creation
vlan 10
name HR
vlan 20
name IT
vlan 30
name FINANCE
vlan 40
name WIFI

! Enable Layer 3 Routing
ip routing

! SVI Interfaces (Gateways)
interface vlan 10
ip address 192.168.10.1 255.255.255.0
no shutdown

interface vlan 20
ip address 192.168.20.1 255.255.255.0
no shutdown

interface vlan 30
ip address 192.168.30.1 255.255.255.0
no shutdown

interface vlan 40
ip address 192.168.40.1 255.255.255.0
no shutdown

! Trunk to Access Switch
interface f0/1
switchport mode trunk

! Wireless AP Port
interface fa0/2
switchport mode access
switchport access vlan 40

! DHCP Configuration
ip dhcp pool HR
network 192.168.10.0 255.255.255.0
default-router 192.168.10.1

ip dhcp pool IT
network 192.168.20.0 255.255.255.0
default-router 192.168.20.1

ip dhcp pool FINANCE
network 192.168.30.0 255.255.255.0
default-router 192.168.30.1

ip dhcp pool WIFI
network 192.168.40.0 255.255.255.0
default-router 192.168.40.1

! ACL Configuration
ip access-list extended BLOCK_HR
deny ip 192.168.10.0 0.0.0.255 192.168.30.0 0.0.0.255
permit ip any any

! Apply ACL to VLAN 10
interface vlan 10
ip access-group BLOCK_HR in

! STP Optimization
spanning-tree vlan 10,20,30,40 priority 4096

end
write memory



📄 ACCESS SWITCH CONFIG (L2 SWITCH)


enable
configure terminal

! VLAN Creation
vlan 10
name HR
vlan 20
name IT
vlan 30
name FINANCE


! Assign Ports to VLANs
interface range fa0/2-3
switchport mode access
switchport access vlan 10

interface range fa0/4-5
switchport mode access
switchport access vlan 20

interface range fa0/6-7
switchport mode access
switchport access vlan 30

! Trunk to Core Switch
interface f0/1
switchport mode trunk

! Port Security
interface range fa0/2-7
switchport port-security
switchport port-security maximum 1
switchport port-security mac-address sticky
switchport port-security violation restrict

end
write memory
